// Widget autonome "bouton play + timing + recherche + tableau", pour

// mapRow convertit chaque ligne Arrow renvoyée par db.query en objet JS
// simple ; valueField est la colonne numérique utilisée pour la part (%)
// affichée dans la colonne "part". extraBar ajoute une 2e colonne barre
// (une autre couleur) sur un second champ numérique, pour comparer d'un
// coup d'oeil deux proportions (ex : part des parcelles vs part de la
// surface).
function liveQueryDemo({
  sql,
  mapRow,
  valueField,
  searchFields,
  columns,
  header,
  width,
  valueFormat = frInt,
  searchPlaceholder = "🔍 Rechercher…",
  buttonLabel = "Tester en live",
  barColor = "#4758AB",
  extraBar = null,
  format = {},
  align = {}
}) {
  const btn = runLiveButton(buttonLabel);
  const search = searchInput(searchPlaceholder);
  search.style.display = "none";
  const statusBox = html`<div></div>`;
  const resultsBox = html`<div></div>`;
  let currentRows = null;

  function renderStatus(elapsed_ms) {
    statusBox.replaceChildren(
      elapsed_ms == null
        ? html`<p class="testlive-placeholder">Lancer la démo live.</p>`
        : html`<p class="testlive-timing">⏱️ Temps d'exécution : <strong>${elapsed_ms.toFixed(0)} ms</strong> · ${totalInputRows.toLocaleString("fr-FR")} → ${currentRows.length} lignes</p>`
    );
  }

  function renderTable() {
    const q = search.value.trim().toLowerCase();
    const rows = !q
      ? currentRows
      : currentRows.filter((r) => searchFields.map((f) => r[f]).join(" ").toLowerCase().includes(q));
    const extraFormat = extraBar
      ? { [extraBar.key]: (v) => barCell(v, extraBar.color ?? barColor) }
      : {};
    resultsBox.replaceChildren(
      html`<div class="testlive-results">${Inputs.table(rows, {
        columns,
        header,
        width,
        align: { [valueField]: "right", ...align },
        format: { [valueField]: valueFormat, part: (v) => barCell(v, barColor), ...extraFormat, ...format },
        layout: "fixed",
        rows: 10
      })}</div>`
    );
  }

  renderStatus(null);

  search.addEventListener("input", renderTable);

  btn.addEventListener("input", async () => {
    const t0 = performance.now();
    const raw = await db.query(sql);
    const rows = Array.from(raw, mapRow);
    const total = d3.sum(rows, (d) => d[valueField]);
    let withParts = rows.map((d) => ({ ...d, part: total > 0 ? d[valueField] / total : 0 }));
    if (extraBar) {
      const totalExtra = d3.sum(rows, (d) => d[extraBar.field]);
      withParts = withParts.map((d) => ({
        ...d,
        [extraBar.key]: totalExtra > 0 ? d[extraBar.field] / totalExtra : 0
      }));
    }
    currentRows = withParts;
    renderStatus(performance.now() - t0);
    search.value = "";
    search.style.display = "";
    renderTable();
  });

  return html`<div class="testlive-widget">${btn}${statusBox}${search}${resultsBox}</div>`;
}

SELECT
    COALESCE(d.nom_chapitre, 'Non classé') AS famille,
    COUNT(*) AS n_parcelles,
    SUM(ST_Area(r.geom)) / 10000 AS surface_ha
FROM read_parquet('${RPG_URL}') r
LEFT JOIN dico d ON r.code_cultu = d.code_culture
GROUP BY 1
ORDER BY surface_ha DESC

// Parcourt récursivement un tableau de coordonnées GeoJSON (Polygon ou
// MultiPolygon) pour calculer une bounding box, sans dépendance à turf.
function geojsonBounds(fc) {
  let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
  function visit(coords) {
    if (typeof coords[0] === "number") {
      const [x, y] = coords;
      if (x < minX) minX = x;
      if (x > maxX) maxX = x;
      if (y < minY) minY = y;
      if (y > maxY) maxY = y;
    } else {
      coords.forEach(visit);
    }
  }
  fc.features.forEach((f) => visit(f.geometry.coordinates));
  return [[minX, minY], [maxX, maxY]];
}

SELECT

  // Le timing affiché ici est celui du dernier surlignage (updateAllMapHighlight),
  // pas celui du chargement des données : c'est la partie "fluide" qu'on veut
  // montrer — changer de culture ne relance ni fetch ni requête.
  return html`<p class="testlive-timing">⏱️ Surlignage recalculé en <strong>${updateAllMapHighlight.toFixed(2)} ms</strong> · ${allParcelsData.rows.length.toLocaleString("fr-FR")} parcelles en mémoire</p>`;
}
viewof cultureSel = Inputs.select(cultureList, { label: "Culture à surligner", value: "Blé tendre" })
updateAllMapHighlight = {
  await mapAllReady;
  if (allParcelsData == null) return 0;
  const t0 = performance.now();
  const isSelected = cultureSel === AUTRES_LABEL
    ? ["!", ["in", ["get", "nom_culture"], ["literal", topCultures]]]
    : ["==", ["get", "nom_culture"], cultureSel];
  mapAll.setPaintProperty("parcels-all-fill", "fill-color", ["case", isSelected, "#ec6842", "#c9c9c2"]);
  mapAll.setPaintProperty("parcels-all-fill", "fill-opacity", ["case", isSelected, 0.85, 0.15]);
  mapAll.setPaintProperty("parcels-all-line", "line-color", ["case", isSelected, "#7a2f10", "#20201d"]);
  mapAll.setPaintProperty("parcels-all-line", "line-opacity", ["case", isSelected, 0.9, 0.15]);
  return performance.now() - t0;
}
selectedCultureStats = {
  updateAllMapHighlight;
  if (allParcelsData == null) return null;
  const rows = allParcelsData.rows;
  const totalSurface = d3.sum(rows, (d) => d.surface_ha);
  const selRows = cultureSel === AUTRES_LABEL
    ? rows.filter((d) => !topCultureSet.has(d.nom_culture))
    : rows.filter((d) => d.nom_culture === cultureSel);
  const surface = d3.sum(selRows, (d) => d.surface_ha);
  return { n: selRows.length, surface, part: totalSurface > 0 ? surface / totalSurface : 0 };
}
{
  if (selectedCultureStats == null) {
    return html`<p class="testlive-placeholder">En attente du chargement de la carte…</p>`;
  }
  const s = selectedCultureStats;
  return html`<div class="highlight-stats">
    <div class="stat-value">${s.n.toLocaleString("fr-FR")}</div>
    <div class="stat-label">parcelles de « ${cultureSel} »</div>
    <div class="stat-value">${frNumber(s.surface, 0)} ha</div>
    <div class="stat-label">soit ${(s.part * 100).toFixed(1)} % de la surface du département</div>
  </div>`;
}

Pour reproduire chez soi

C’est compliqué à reproduire chez soi ?

Les outils présentés ici sont tous open source, utilisables de nombreuses manières !

Parquet est un format de fichier lisible par de nombreux logiciels (comme CSV mais optimisé pour la data)

DuckDB est le logiciel le plus puissant pour lire un Parquet.

Il peut s’utiliser directement ou via , Python () et même QGIS !

Mais on est obligé d’avoir un cloud ?

Même sur un ordinateur personnel ça marche. On peut lire une donnée

  • depuis son disque dur
  • ou depuis internet (plein de Parquet sur data.gouv)

Avec le cloud on pourra aller plus loin grâce à des optimisations supplémentaires.
